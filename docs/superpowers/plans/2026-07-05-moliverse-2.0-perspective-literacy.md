# MoliVerse 2.0 — Perspective Literacy Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild MoliVerse as a focused Perspective-Literacy app: search an event → compare how outlets headline it → read *why* the framing differs → see verified facts → reflect.

**Architecture:** Static Next.js 14 App Router site. All content is curated, typed TypeScript data (no backend, no LLM, no network). Landing page filters events client-side; each event is a statically-generated editorial "dossier" page. Reflection answers persist to `localStorage`.

**Tech Stack:** Next.js 14 (App Router), React 18, TypeScript, Tailwind CSS 3, `next/font/google` (Source Serif 4 + Inter), Vitest for unit tests.

## Global Constraints

- No external network calls, no API keys, no `/api` routes at runtime. All data is in-bundle.
- Visual identity is **Editorial Newsprint**: paper `#f5f1e8`, ink `#1a1a1a`, red accent `#c0392b`, verified-green `#1a7a4a`. Serif for headings/body; sans (Inter) only for small labels/chips/tags.
- Path alias `@/*` maps to repo root (already configured in `tsconfig.json`).
- Server components by default; only components needing browser APIs use `"use client"`.
- Ship exactly **3 event dossiers**, each with **4 sources**.
- Every `contestedWords` string MUST be an exact substring of its source `headline`.
- Frequent commits: one per task minimum, following the steps.

- [ ] **Step 3: Create `lib/events/validate.ts`**

```ts
import { Event } from "@/lib/types";

export function validateEvent(e: Event): string[] {
  const problems: string[] = [];
  if (!e.slug) problems.push("missing slug");
  if (!e.title) problems.push("missing title");
  if (e.sources.length < 2) problems.push("needs at least 2 sources");
  if (e.contestedTerms.length < 2) problems.push("needs at least 2 contested terms");
  if (e.backgroundFacts.length < 1) problems.push("needs at least 1 background fact");
  if (e.reflectionQuestions.length < 1) problems.push("needs at least 1 reflection question");

  const terms = new Set(e.contestedTerms.map((t) => t.term));
  for (const s of e.sources) {
    if (!terms.has(s.framingWord)) {
      problems.push(`source ${s.outlet}: framingWord "${s.framingWord}" has no contested term`);
    }
    for (const w of s.contestedWords) {
      if (!s.headline.includes(w)) {
        problems.push(`source ${s.outlet}: "${w}" not in headline`);
      }
    }
  }
  return problems;
}
```

import { Fragment } from "react";

function escapeRegExp(s: string): string {
  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}

export default function Highlight({
  text,
  marks,
  accent = "#c0392b",
}: {
  text: string;
  marks: string[];
  accent?: string;
}) {
  const valid = marks.filter((m) => m && text.includes(m));
  if (valid.length === 0) return <>{text}</>;

  const pattern = new RegExp(`(${valid.map(escapeRegExp).join("|")})`, "g");
  const parts = text.split(pattern);

  return (
    <>
      {parts.map((part, i) =>
        valid.includes(part) ? (
          <mark
            key={i}
            className="rounded-[2px] px-0.5"
            style={{ backgroundColor: `${accent}22`, color: "inherit" }}
          >
            {part}
          </mark>
        ) : (
          <Fragment key={i}>{part}</Fragment>
        )
      )}
    </>
  );
}

import { Event } from "@/lib/types";

export function searchEvents(events: Event[], query: string): Event[] {
  const q = query.trim().toLowerCase();
  if (!q) return events;

  return events.filter((e) => {
    const haystack = [
      e.title,
      e.category,
      ...e.aliases,
      ...e.sources.map((s) => s.outlet),
    ]
      .join(" ")
      .toLowerCase();
    return haystack.includes(q);
  });
}

"use client";

import { useEffect, useState } from "react";

function keyFor(slug: string, i: number) {
  return `moliverse:reflection:${slug}:${i}`;
}

function readStored(slug: string, i: number): string {
  try {
    return localStorage.getItem(keyFor(slug, i)) ?? "";
  } catch {
    return "";
  }
}

export default function ReflectionPrompts({
  slug,
  questions,
}: {
  slug: string;
  questions: string[];
}) {
  const [answers, setAnswers] = useState<string[]>(() => questions.map(() => ""));

  useEffect(() => {
    setAnswers(questions.map((_, i) => readStored(slug, i)));
  }, [slug, questions]);

  function update(i: number, value: string) {
    setAnswers((prev) => {
      const next = [...prev];
      next[i] = value;
      return next;
    });
    try {
      localStorage.setItem(keyFor(slug, i), value);
    } catch {
      /* storage unavailable — persist silently degrades */
    }
  }

  return (
    <div className="space-y-5">
      {questions.map((q, i) => (
        <div key={i} className="border-l-[3px] border-ink pl-4">
          <p className="font-serif text-base italic text-ink">{q}</p>
          <textarea
            value={answers[i] ?? ""}
            onChange={(e) => update(i, e.target.value)}
            placeholder="Your answer…"
            rows={2}
            className="mt-2 w-full rounded-sm border border-rule bg-paper-card px-3 py-2 font-serif text-sm text-ink focus:border-accent focus:outline-none"
          />
        </div>
      ))}
    </div>
  );
}
