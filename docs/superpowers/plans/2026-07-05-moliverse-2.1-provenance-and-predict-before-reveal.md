# MoliVerse 2.1 — Provenance & Predict-Before-Reveal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add verifiable per-source provenance and a predict-before-reveal challenge to the existing three MoliVerse dossiers.

**Architecture:** Content stays static typed data; pages stay statically rendered. Task 1 lands the atomic data-model change (new `Source` fields, cited `BackgroundFact[]`, `Event.methodology`) across types, validation, and all three data files. Tasks 2–3 add provenance UI. Tasks 4–6 add the challenge: a pure helper module, a per-headline guess component, and a client wrapper that replaces the passive headline grid.

**Tech Stack:** Next.js 14 (App Router, RSC), React 18, TypeScript, Tailwind, Vitest + @testing-library/react. Path alias `@/` → repo root.

## Global Constraints

- **Framework floor:** Next.js `14.2.5`, React `^18.3.1`. No new runtime dependencies.
- **No network at request time:** static data only; `url`/`archiveUrl` are plain outbound `<a>` links followed on click. No fetch, no API routes, no LLM.
- **Server-by-default:** only `PerspectiveChallenge` and `HeadlineGuess` are new client components (`"use client"`). Everything else stays a server component.
- **Editorial theme:** reuse existing Tailwind tokens (`paper`, `paper-card`, `ink`, `ink-muted`, `ink-soft`, `rule`, `accent`, `verified`, `verified-bg`, `verified-border`, `kicker`, `prose-news`, `font-serif`, `font-sans`). Do not introduce new colors.
- **Link integrity:** every `archiveUrl` must resolve to a snapshot showing that exact headline. Unverifiable headlines are replaced with a verifiable one, never faked.
- **Test command:** `npx vitest run <path>` for one file; `npm test` for the whole suite.
- **Tone:** wrong guesses are met with insight, never scolding.

Create `lib/challenge.ts`:

```ts
import { Orientation, Source } from "@/lib/types";

export interface HeadlineToken {
  text: string;
  contested: boolean;
}

function escapeRegExp(s: string): string {
  return s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
}

export function tokenizeHeadline(
  headline: string,
  contestedWords: string[],
): HeadlineToken[] {
  const valid = [...new Set(contestedWords.filter((m) => m && headline.includes(m)))].sort(
    (a, b) => b.length - a.length,
  );
  const tokens: HeadlineToken[] = [];
  if (valid.length === 0) {
    for (const word of headline.split(/\s+/).filter(Boolean)) {
      tokens.push({ text: word, contested: false });
    }
    return tokens;
  }
  const validSet = new Set(valid);
  const pattern = new RegExp(`(${valid.map(escapeRegExp).join("|")})`, "g");
  for (const part of headline.split(pattern)) {
    if (!part) continue;
    if (validSet.has(part)) {
      tokens.push({ text: part, contested: true });
    } else {
      for (const word of part.split(/\s+/).filter(Boolean)) {
        tokens.push({ text: word, contested: false });
      }
    }
  }
  return tokens;
}

const ALL_ORIENTATIONS: Orientation[] = [
  "Public broadcaster", "State-owned", "Private / commercial",
  "Left-leaning", "Right-leaning", "Centrist", "Pan-Arab", "Other",
];

export function orientationOptions(sources: Source[]): Orientation[] {
  const distinct = [...new Set(sources.map((s) => s.orientation))];
  const options = [...distinct];
  for (const o of ALL_ORIENTATIONS) {
    if (options.length >= 4) break;
    if (!options.includes(o)) options.push(o);
  }
  for (let i = options.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [options[i], options[j]] = [options[j], options[i]];
  }
  return options;
}
```

```tsx
"use client";

import { useState } from "react";
import SourceHeadline from "@/components/SourceHeadline";
import { tokenizeHeadline } from "@/lib/challenge";
import { Orientation, Source } from "@/lib/types";

export default function HeadlineGuess({
  source,
  orientationOptions,
  revealed,
  onReveal,
}: {
  source: Source;
  orientationOptions: Orientation[];
  revealed: boolean;
  onReveal: () => void;
}) {
  const [tapped, setTapped] = useState<number | null>(null);
  const [picked, setPicked] = useState<Orientation | null>(null);
  const tokens = tokenizeHeadline(source.headline, source.contestedWords);

  if (revealed) {
    return (
      <div>
        <SourceHeadline source={source} />
        <p className="mt-2 font-serif text-[13px] italic text-ink-soft">
          {source.insight}
        </p>
      </div>
    );
  }

  const wordCorrect = tapped !== null && tokens[tapped]?.contested;
  const orientationCorrect = picked !== null && picked === source.orientation;

  return (
    <div
      className="rounded-sm border border-rule bg-paper-card p-4"
      style={{ borderTopWidth: 3, borderTopColor: "#999" }}
    >
      <div className="font-sans text-[11px] uppercase tracking-wide text-ink-muted">
        Mystery outlet · guess before you reveal
      </div>
      <p className="mt-2 font-serif text-[15px] leading-snug text-ink">
        {tokens.map((t, i) => (
          <button
            key={i}
            type="button"
            onClick={() => setTapped(i)}
            className={`rounded-[2px] px-0.5 ${
              tapped === i
                ? t.contested
                  ? "bg-verified-bg text-verified"
                  : "bg-paper line-through"
                : "hover:bg-paper"
            }`}
          >
            {t.text}
          </button>
        ))}
      </p>
      {tapped !== null && (
        <p className="mt-1 font-sans text-[11px] text-ink-muted">
          {wordCorrect
            ? "That's a framing word — it does persuading work."
            : "Look again — which word carries the judgment?"}
        </p>
      )}

      <div className="mt-3 font-sans text-[11px] text-ink-muted">Who wrote this?</div>
      <div className="mt-1 flex flex-wrap gap-1.5">
        {orientationOptions.map((o) => (
          <button
            key={o}
            type="button"
            onClick={() => setPicked(o)}
            className={`rounded-full border px-2 py-0.5 font-sans text-[11px] ${
              picked === o
                ? o === source.orientation
                  ? "border-verified text-verified"
                  : "border-accent text-accent"
                : "border-rule text-ink-soft hover:border-ink"
            }`}
          >
            {o}
          </button>
        ))}
      </div>
      {picked !== null && (
        <p className="mt-1 font-sans text-[11px] text-ink-muted">
          {orientationCorrect ? "Good read." : "A common guess — reveal to see who it really was."}
        </p>
      )}

      <button
        type="button"
        onClick={onReveal}
        className="mt-3 font-sans text-[12px] text-accent hover:underline"
      >
        Reveal outlet →
      </button>
    </div>
  );
}
```

```tsx
"use client";

import { useMemo, useState } from "react";
import HeadlineGuess from "@/components/HeadlineGuess";
import { orientationOptions } from "@/lib/challenge";
import { Source } from "@/lib/types";

export default function PerspectiveChallenge({ sources }: { sources: Source[] }) {
  const [revealed, setRevealed] = useState<boolean[]>(() => sources.map(() => false));
  const options = useMemo(() => orientationOptions(sources), [sources]);
  const allRevealed = revealed.every(Boolean);

  function reveal(index: number) {
    setRevealed((prev) => prev.map((r, i) => (i === index ? true : r)));
  }
  function revealAll() {
    setRevealed(sources.map(() => true));
  }

  return (
    <div>
      <div className="grid gap-3 sm:grid-cols-2">
        {sources.map((s, i) => (
          <HeadlineGuess
            key={s.outlet}
            source={s}
            orientationOptions={options}
            revealed={revealed[i]}
            onReveal={() => reveal(i)}
          />
        ))}
      </div>
      {!allRevealed && (
        <button
          type="button"
          onClick={revealAll}
          className="mt-3 font-sans text-[12px] text-ink-muted hover:text-accent hover:underline"
        >
          Just let me read →
        </button>
      )}
    </div>
  );
}
```
